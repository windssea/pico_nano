set(z_sources vendor/adler32.c vendor/crc32.c vendor/inffast.c vendor/inflate.c vendor/inftrees.c vendor/zutil.c)
if(ESP_PLATFORM)
    idf_component_register(SRCS ${z_sources} INCLUDE_DIRS vendor)
    set(z_target ${COMPONENT_LIB})
else()
    add_library(pn_zlib STATIC ${z_sources})
    set(z_target pn_zlib)
    target_include_directories(${z_target} PUBLIC vendor)
    if(PN_SANITIZERS)
        target_compile_options(${z_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${z_target} PUBLIC -fsanitize=address,undefined)
    endif()
endif()
target_compile_definitions(${z_target} PUBLIC Z_SOLO Z_PREFIX)
