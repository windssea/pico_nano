set(expat_sources vendor/lib/xmlparse.c vendor/lib/xmlrole.c vendor/lib/xmltok.c vendor/lib/xcs.c)
if(ESP_PLATFORM)
    idf_component_register(SRCS ${expat_sources} INCLUDE_DIRS vendor/lib PRIV_INCLUDE_DIRS config)
    set(expat_target ${COMPONENT_LIB})
else()
    add_library(pn_expat STATIC ${expat_sources})
    set(expat_target pn_expat)
    target_include_directories(${expat_target} PUBLIC vendor/lib PRIVATE config)
    if(PN_SANITIZERS)
        target_compile_options(${expat_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${expat_target} PUBLIC -fsanitize=address,undefined)
    endif()
endif()
target_compile_definitions(${expat_target} PUBLIC XML_STATIC)
